Step 1: Filter the data to include only rows where the issue price is $34.95.  
Step 2: Extract the 'total mintage' values for these rows in chronological order (by year).  
Step 3: Calculate the percentage increase between consecutive years using the formula:  
        ((mintage_next - mintage_current) / mintage_current) * 100  
Step 4: Compute the average of these percentage increases.  
Step 5: Print the result.

```python
import pandas as pd

df = pd.read_csv('table.csv')
# Filter rows where issue price is $34.95
filtered_df = df[df['issue price'] == '34.95'].copy()
# Convert total mintage to integer for calculation
filtered_df['total mintage'] = filtered_df['total mintage'].astype(int)
# Sort by year to ensure chronological order
filtered_df = filtered_df.sort_values('year')
# Calculate percentage increase between consecutive years
percentage_increases = []
for i in range(1, len(filtered_df)):
    current = filtered_df.iloc[i-1]['total mintage']
    next_val = filtered_df.iloc[i]['total mintage']
    increase = ((next_val - current) / current) * 100
    percentage_increases.append(increase)
# Calculate average percentage increase
avg_increase = sum(percentage_increases) / len(percentage_increases)
print(f"Final Answer: {avg_increase:.2f}")
```
