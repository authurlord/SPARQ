```python
import pandas as pd

df = pd.read_csv('table.csv')

# Filter top 5 ranked countries (rank 1 to 5)
top_5 = df[df['rank'].astype(float).between(1, 5)]

# Convert relevant columns to numeric
df_numeric = top_5[['2009', '2010', '2011']].apply(pd.to_numeric, errors='coerce')

# Compute growth rates
growth_2009_to_2010 = ((df_numeric['2010'] - df_numeric['2009']) / df_numeric['2009']) * 100
growth_2010_to_2011 = ((df_numeric['2011'] - df_numeric['2010']) / df_numeric['2010']) * 100

# Average growth rate per country
avg_growth_2009_to_2010 = growth_2009_to_2010.mean()
avg_growth_2010_to_2011 = growth_2010_to_2011.mean()

# Percentage difference between the two average growth rates
percentage_difference = (avg_growth_2010_to_2011 - avg_growth_2009_to_2010) * 100

print(f"Final Answer: {percentage_difference:.2f}")
```
