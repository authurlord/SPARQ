import pandas as pd

df = pd.read_csv('table.csv')

# Describe the main columns
print("Column Descriptions:")
print("position: Rank of the driver in the standings.")
print("driver: Name of the racing driver.")
print("points: Total points earned by the driver.")
print("starts: Number of races started by the driver.")
print("wins: Number of race wins.")
print("top 5s: Number of times the driver finished in the top 5.")
print("top 10s: Number of times the driver finished in the top 10.")
print("winnings: Total prize money earned by the driver.")

# Initial observations
print("\nInitial Observations:")
print(f"Total number of drivers: {len(df)}")
print(f"Number of races started by all drivers: {df['starts'].sum()}")
print(f"Average points: {df['points'].mean():.1f}")
print(f"Most wins: {df['wins'].max()} (by Andrew Ranger)")
print(f"Highest winnings: ${df['winnings'].max():,}")

# Summary statistics for numerical columns
print("\nSummary Statistics (Numerical Columns):")
print(df[['points', 'wins', 'top 5s', 'top 10s', 'winnings']].describe())

# Check for ties in points
points_tied = df[df['points'] == df['points'].max()]
if len(points_tied) > 1:
    print(f"\nTie in points: {len(points_tied)} drivers have the highest points ({df['points'].max()}).")
else:
    print(f"\nNo tie in points. Andrew Ranger leads with {df['points'].max()} points.")

print(f"Final Answer: position, driver, points, starts, wins, top 5s, top 10s, winnings")
