import pandas as pd
import matplotlib.pyplot as plt

# Load the data
df = pd.read_csv('table.csv')

# Extract viewership data (in millions) and years
viewers = df['Viewers'].dropna().astype(str).str.replace(' million', '').astype(float)
years = df['Year'].dropna()

# Create a waterfall chart
plt.figure(figsize=(10, 6))
plt.title('Trend in Viewership (in millions) from 2007 to 2013')
plt.xlabel('Year')
plt.ylabel('Viewers (millions)')

# Calculate the changes for waterfall effect
changes = [viewers.iloc[0]] + [viewers.iloc[i] - viewers.iloc[i-1] for i in range(1, len(viewers))]

# Plot the bars
colors = ['green' if x >= 0 else 'red' for x in changes]
plt.bar(years, changes, color=colors, edgecolor='black')

# Annotate values
for i, v in enumerate(changes):
    plt.text(i, v + 0.1, f'{v:.1f}', ha='center', va='bottom', fontsize=9)

# Show the plot
plt.tight_layout()
plt.show()

# Final answer is the list of viewership numbers
print("Final Answer: 6.574, 6.668, 6.487, 5.709, 6.337, 5.1, 5.5")
